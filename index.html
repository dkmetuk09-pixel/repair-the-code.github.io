const addButtons = document.querySelectorAll('.add-btn'); const cartCountEl = document.getElementById('cartCount'); const cartBtn = document.getElementById('cartBtn'); const yearEl = document.getElementById('year'); yearEl.textContent = new Date().getFullYear(); let cart = {}; function updateCartUI(){ const totalItems = Object.values(cart).reduce((s,p)=>s+p.qty,0); cartCountEl.textContent = totalItems; cartBtn.animate([{transform:'scale(1)'},{transform:'scale(1.08)'},{transform:'scale(1)'}],{duration:250}); } addButtons.forEach(btn=>{ btn.addEventListener('click', (e)=>{ const card = e.target.closest('[data-id]'); const id = card.dataset.id; const name = card.dataset.name; const price = parseFloat(card.dataset.price); if (!cart[id]) cart[id] = {name, price, qty:0}; cart[id].qty += 1; updateCartUI(); }); }); cartBtn.addEventListener('click', ()=>{ const items = Object.values(cart); if (items.length===0) { alert('Кошик порожній.'); return; } let msg = 'Ваш кошик: '; let total = 0; items.forEach(i=>{ msg += `${i.name} × ${i.qty} — €${(i.price*i.qty).toFixed(2)} `; total += i.price*i.qty; }); msg += ` Загалом: €${total.toFixed(2)} Це демо: оформлення замовлення не реалізовано.`; alert(msg); }); })();
